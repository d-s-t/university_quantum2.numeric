\subsection{תנאי פחות מחמיר עבור מציאת אנרגיה קשורה}
בסעיפים הקודמים מצאנו את האנרגיה הקשורה ע"י התנאי $u(R\mid E) = 0$. זו דרישה מחמירה כי זה נכון עבור $R\rightarrow\infty$, ניתן גם לומר שזה קירוב בסדר 0 למה שאנחנו רוצים. כדי להגיע לדיוק יותר גבוה נרצה תנאי יותר חלש על $u(R\mid E)$ כך שיתאים גם לערכי R קטנים יותר.\\
נחזור למשוואה הדפרנציאלית שלנו:
$$u''+W(r)u=0$$
ונשים לב כי עבור $r\gg a_B$ הפונקציה W נראית כמו:
$$W=\frac{2\mu}{\hbar^2}E = -k^2$$
ואז המשוואה נהיית פשוט 
$u''=k^2u$
שהפתרוות שלה הם $e^{kr},e^{-kr}$. ברור כי הפתרון $e^{kr}$ מתאים לתוצאות שאנחנו לא רוצים (אלה שמתבדרות), ולכן הפתרון הרלוונטי הוא 
$$u(r\gg a_B) = Ae^{-kr}\,,\qquad k = \sqrt{-\frac{2\mu}{\hbar^2}E}$$
(נזכור כי $E<0$ שהרי אנחנו מעוניינים באנרגיות קשורות ולא של חלקיק חופשי)\\
כלומר במקום לדרוש $u(R)=0$ אנחנו יכולים לדרוש ש־$u(r\gg a_B)$ מתנהג כמו $Ae^{-kr}$
נשים לב כי 
$$ln|u(r\gg a_B)| = ln|A|-kr \Rightarrow \frac{d}{dr}(ln|u|) = -k$$



\section{קינמטיקה יחסותית}
\subsection{}
\begin{align*}
    \symbfit{p}^2 c^2 + m^2 c^4 - {\left(E + mc^2-V_c\right)}^2 &= \symbfit{p}^2 c^2 + m^2 c^4 - {\left(E ^2+ \left(mc^2\right)^2+V_c^2 + 2Emc^2 - 2EV_c -2V_cmc^2\right)} \\ 
    &\indent\indent= \symbfit{p}^2 c^2 - \left(E ^2 + V_c^2 + 2Emc^2 - 2EV_c -2V_cmc^2\right)
\end{align*}
נחלק את הביטוי ב$2mc^2$ ונקבל
$$\frac{\symbfit{p}^2}{2m} - \left(E - V_c\right) - \left(\frac{E}{2}\frac{E}{mc^2} + \frac{V_c}{2}\frac{V_c}{mc^2} - V_c\frac{E}{mc^2} \right)$$
נזרוק את האיברים מהצורה $\frac{E}{mc^2}$ ו־$\frac{V_c}{mc^2}$
ונקבל את הביטוי
$$ \frac{\symbfit{p}^2}{2m} - \left(E - V_c \right)$$
\subsection{}
משוואה 2:
$$\psi(\symbfit{r}) = \frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)$$
כמובן שאופרטור התנע הוא $\symbfit{p}=i\hbar\nabla$ ולכן
$\symbfit{p}^2 = -\hbar^2\nabla^2$. 
נציב:
\begin{align*}
    &\left[\symbfit{p}^2 c^2 + m^2 c^4 - {\left(E + mc^2-V_c\right)}^2\right]\psi= \\ 
    &\indent\indent=\left[-\hbar^2c^2\nabla^2 + m^2 c^4 - {\left(E + mc^2-V_c\right)}^2\right]\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right)=\\
    &\indent\indent=-\hbar^2c^2\nabla^2\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right) + \left[m^2 c^4 - {\left(E + mc^2-V_c\right)}^2\right]\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right) = 0
\end{align*}
נפתחת את הלפלסיאן בקואורדינטות כדוריות:
$$\Delta f=\frac{1}{r^{2}}\frac{\partial}{\partial r}\left(r^{2}\frac{\partial f}{\partial r}\right)+\frac{1}{r^{2}\sin \theta }\frac{\partial}{\partial \theta }\left(\sin \theta \frac{\partial f}{\partial \theta }\right)+\frac{1}{r^{2}\sin ^{2}\theta }\frac{\partial ^{2}f}{\partial \phi ^{2}}$$
\begin{align*}
    \nabla^2\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right) &= \frac{1}{r^{2}}\frac{\partial}{\partial r}\left(r^{2}\frac{\partial}{\partial r}\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right)\right) +\frac{1}{r^{2}\sin \theta }\frac{\partial}{\partial \theta }\left(\sin \theta \frac{\partial}{\partial \theta }\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right)\right) +\frac{1}{r^{2}\sin ^{2}\theta }\frac{\partial ^{2}}{\partial \phi ^{2}}\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right) \\
    &= \frac{Y_{lm}(\hat{\symbfit{r}})}{r^{2}}\frac{\partial}{\partial r}\left(r^{2}\frac{\partial}{\partial r}\left(\frac{u(r)}{r}\right)\right) +\frac{u(r)}{r^{3}\sin \theta }\frac{\partial}{\partial \theta }\left(\sin \theta \frac{\partial}{\partial \theta }\left(Y_{lm}(\hat{\symbfit{r}})\right)\right) +\frac{u(r)}{r^{3}\sin ^{2}\theta }\frac{\partial ^{2}}{\partial \phi ^{2}}\left(Y_{lm}(\hat{\symbfit{r}})\right) 
\end{align*}

כאשר הלפלסיאן פועל על $Y_{lm}$ מתקבל $l(l+1)Y_{lm}$ ולכן

\begin{align*}
\nabla^2\left(\frac{1}{r}Y_{lm}(\hat{\symbfit{r}})u(r)\right)
    &= \frac{Y_{lm}(\hat{\symbfit{r}})}{r^{2}}\frac{\partial}{\partial r}\left(r^{2}\left(\frac{u'(r)}{r}-\frac{u(r)}{r^2}\right)\right) + \frac{u(r)}{r^3}l(l+1)Y_{lm}(\hat{\symbfit{r}})\\
    &= \frac{Y_{lm}(\hat{\symbfit{r}})}{r^{2}}\frac{\partial}{\partial r}\left(ru'(r)-u(r)\right) + \frac{u(r)}{r^3}l(l+1)Y_{lm}(\hat{\symbfit{r}})\\
    &= \frac{Y_{lm}(\hat{\symbfit{r}})}{r^{2}}\left(u'(r)+ru''(r)-u'(r)\right) + \frac{u(r)}{r^3}l(l+1)Y_{lm}(\hat{\symbfit{r}})\\
    &= \frac{Y_{lm}(\hat{\symbfit{r}})}{r}\left(u''(r) + \frac{u(r)}{r^2}l(l+1)\right)
\end{align*}
נציב את זה חזרה בשוואה מעלה, ונחלק ב־$-\hbar^2c^2\frac{Y_{lm}}{r}$
\begin{align*}
    &u''(r) + \frac{u(r)}{r^2}l(l+1) - \frac{m^2 c^4 - {\left(E + mc^2-V_c\right)}^2}{\hbar^2c^2}u(r) = \\
    =&\ u''(r) + \left(\frac{{\left(E + mc^2-V_c\right)}^2 - m^2 c^4}{\hbar^2c^2} + \frac{l(l+1)}{r^2}\right)u(r) = 0
\end{align*}
ולכן
$$W(r)=\frac{{\left(E + mc^2-V_c\right)}^2 - m^2 c^4}{\hbar^2c^2} + \frac{l(l+1)}{r^2}$$
